# Agent UI — Session 3: Empty States + Mobile + Contextual Follow-ups

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the bare "Send a message to start" empty state into a role-aware set of clickable starter prompts, fix two mobile layout bugs (code block overflow + agent bubble cramp), unify the mobile and desktop conversation-list render paths so they can't drift, and wire a contextual follow-up helper that replaces the hardcoded "Tell me more / What should I focus on next?" generics with 3 topical suggestions generated by a fast Haiku call.

**Architecture:** A new `starter_prompts` text array column on `agent_employees` lets each employee carry a curated set of conversation starters. The AgentChat empty-state branch renders them as pills that send the message directly. Mobile CSS gets targeted additions — `overflow-x: auto` with a visible scrollbar on `.message-content pre`, and a right gutter on `w-full` agent bubbles below `md:`. A shared `ConversationList` component replaces the two divergent list renders in `page.tsx` (MobileConversationPanel) and `sidebar.tsx`. Finally, a new server endpoint `/api/agent/followups` takes a prompt + final response and returns 3 topical follow-ups via Haiku; the web client calls it on stream completion if a feature flag is set, replacing the hardcoded fallback in `getFollowUpSuggestions`.

**Tech Stack:** TypeScript, Drizzle, React 19, Tailwind 4, `@anthropic-ai/sdk` (Haiku model), Playwright for the audit.

**Prerequisites:**
- Sessions 1 + 2 merged and green per `docs/superpowers/audits/agent-ui-session-*.last-run.txt`.
- Session 2.5 prompt caching already shipped (commit `7994235`) — Session 3 audit runs will benefit.
- API on `localhost:3001`, web on `localhost:3000`.
- Anthropic credit available (expect ~$1–2 for the full Session 3 audit after caching savings).

```ts
/**
 * Seed starter prompts onto existing agent employees.
 *
 * Run: pnpm --filter @deft/api exec tsx src/scripts/seed-starter-prompts.ts
 */
import { db } from '../lib/db.js';
import { agentEmployees } from '@deft/db/schema';
import { eq } from 'drizzle-orm';

const PROMPTS_BY_ROLE: Record<string, string[]> = {
  project_manager: [
    "What's overdue?",
    "Who's blocked right now?",
    "Draft today's standup",
    "Summarize last sprint's wins",
    "Which tasks need my attention?",
  ],
  engineering_lead: [
    "What PRs need review?",
    "Any CI failures today?",
    "Summarize this week's merges",
    "Who's pairing on what?",
    "What's in the backlog for next sprint?",
  ],
  executive_assistant: [
    "What's on my calendar today?",
    "Any urgent emails?",
    "Draft a thank-you note",
    "Summarize yesterday's meetings",
    "What do I need to prep for tomorrow?",
  ],
  custom: [
    "What can you help with?",
    "Show me your tools",
    "What do you know about this project?",
  ],
};

async function main() {
  const emps = await db.select().from(agentEmployees);
  console.log(`Found ${emps.length} employees`);
  for (const emp of emps) {
    const prompts = PROMPTS_BY_ROLE[emp.role];
    if (!prompts) {
      console.log(`  skip ${emp.name} (${emp.role}) — no prompt set defined`);
      continue;
    }
    if (emp.starter_prompts && emp.starter_prompts.length > 0) {
      console.log(`  skip ${emp.name} — already has ${emp.starter_prompts.length} prompts`);
      continue;
    }
    await db
      .update(agentEmployees)
      .set({ starter_prompts: prompts })
      .where(eq(agentEmployees.id, emp.id));
    console.log(`  seeded ${emp.name} (${emp.role}) with ${prompts.length} prompts`);
  }
  console.log('Done.');
  process.exit(0);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```tsx
'use client';

import { api } from '@/lib/api';
import Link from 'next/link';
import { X } from 'lucide-react';

export type ConversationListItem = {
  id: string;
  title: string | null;
  updated_at: string;
  agent_employee_id?: string | null;
};

type Props = {
  conversations: ConversationListItem[];
  activeId: string | null;
  /** Per-conversation href builder — passed in so pages can control URL format. */
  hrefFor: (conv: ConversationListItem) => string;
  /** Called after a conversation is deleted (parent can refetch / navigate). */
  onDelete?: (id: string) => void;
  /** Called when a conversation link is clicked (e.g. to close a mobile panel). */
  onNavigate?: () => void;
  emptyText?: string;
  maxItems?: number;
};

export function ConversationList({
  conversations,
  activeId,
  hrefFor,
  onDelete,
  onNavigate,
  emptyText = 'No conversations yet',
  maxItems = 20,
}: Props) {
  const handleDelete = async (id: string, e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();
    await api.delete(`/api/agent/conversations/${id}`);
    onDelete?.(id);
  };

  if (conversations.length === 0) {
    return (
      <p className="text-[12px] text-center py-4" style={{ color: 'var(--outline)' }}>
        {emptyText}
      </p>
    );
  }

  return (
    <div className="flex flex-col gap-0.5">
      {conversations.slice(0, maxItems).map((conv) => {
        const isActive = activeId === conv.id;
        return (
          <Link
            key={conv.id}
            href={hrefFor(conv)}
            onClick={onNavigate}
            className="flex items-center justify-between gap-2 px-2 py-2 rounded-lg text-[0.8125rem] group"
            style={{
              background: isActive ? 'var(--accent-subtle)' : 'transparent',
              color: isActive ? 'var(--accent)' : 'var(--foreground-secondary)',
            }}
          >
            <span className="truncate flex-1" title={conv.title ?? 'Untitled'}>
              {conv.title || 'Untitled'}
            </span>
            {onDelete && (
              <button
                onClick={(e) => handleDelete(conv.id, e)}
                className="p-1 rounded opacity-0 group-hover:opacity-100 group-active:opacity-100 flex-shrink-0"
                style={{ color: 'var(--outline)' }}
                aria-label="Delete conversation"
              >
                <X size={12} />
              </button>
            )}
          </Link>
        );
      })}
    </div>
  );
}
```

```ts
/**
 * POST /api/agent/followups
 * Body: { userPrompt: string; finalResponse: string }
 * Returns: { followups: string[] }
 *
 * Takes a recently-completed user question and agent response and
 * returns 3 short, topical follow-up suggestions generated by a fast
 * Haiku call. Replaces the hardcoded "Tell me more / What should I
 * focus on next?" fallback in the web client.
 *
 * Cost: ~200 tokens per call at Haiku pricing — sub-$0.0005 each.
 */
import { Hono } from 'hono';
import Anthropic from '@anthropic-ai/sdk';
import { env } from '../lib/env.js';
import { z } from 'zod';

const agentFollowupsRoutes = new Hono();

const BodySchema = z.object({
  userPrompt: z.string().min(1).max(4000),
  finalResponse: z.string().min(1).max(20000),
});

agentFollowupsRoutes.post('/', async (c) => {
  const raw = await c.req.json().catch(() => null);
  const parsed = BodySchema.safeParse(raw);
  if (!parsed.success) {
    return c.json({ error: 'Invalid body', code: 'BAD_REQUEST' }, 400);
  }
  if (!env.ANTHROPIC_API_KEY) {
    return c.json({ followups: [] });
  }

  const client = new Anthropic({ apiKey: env.ANTHROPIC_API_KEY });
  const system =
    'You generate 3 SHORT follow-up question suggestions (5-8 words each) that a user might ask an AI assistant next. Return ONLY a JSON array of 3 strings. No markdown, no preamble.';
  const prompt = `The user asked: "${parsed.data.userPrompt}"

The assistant answered: "${parsed.data.finalResponse.slice(0, 1500)}"

What might the user ask next? Return only a JSON array.`;

  try {
    const response = await client.messages.create({
      model: 'claude-haiku-4-5-20251001',
      max_tokens: 200,
      system,
      messages: [{ role: 'user', content: prompt }],
    });

    const textBlock = response.content.find((b) => b.type === 'text');
    if (!textBlock || textBlock.type !== 'text') {
      return c.json({ followups: [] });
    }

    const match = textBlock.text.match(/\[[\s\S]*\]/);
    if (!match) {
      return c.json({ followups: [] });
    }

    const parsedArray = JSON.parse(match[0]);
    if (!Array.isArray(parsedArray)) {
      return c.json({ followups: [] });
    }

    const followups = parsedArray
      .filter((s): s is string => typeof s === 'string')
      .slice(0, 3)
      .map((s) => s.trim())
      .filter((s) => s.length > 0 && s.length < 100);

    return c.json({ followups });
  } catch (err) {
    console.warn('[agent-followups] failed:', err instanceof Error ? err.message : err);
    return c.json({ followups: [] });
  }
});

export { agentFollowupsRoutes };
```
