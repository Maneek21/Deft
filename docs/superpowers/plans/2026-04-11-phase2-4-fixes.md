### Fix 3: Create Agent Employee Trigger Handler

**Files:**
- Create: `apps/api/src/workers/handlers/agent-employee-trigger.ts`
- Modify: `apps/api/src/workers/index.ts`

- [ ] **Step 1: Create the trigger handler**

```typescript
import { eq, and } from 'drizzle-orm';
import { db } from '@deft/db';
import { triggers, agentEmployees, orgs } from '@deft/db/schema';
import { runAgentQuery } from '../../lib/agent-runner.js';

export async function handleAgentEmployeeTrigger(data: {
  triggerId: string;
  orgId: string;
  eventType: string;
  eventData: Record<string, any>;
}) {
  const { triggerId, orgId, eventType, eventData } = data;

  // Load trigger
  const trigger = await db.select().from(triggers)
    .where(and(eq(triggers.id, triggerId), eq(triggers.is_active, true)))
    .limit(1);
  if (!trigger[0]) return;

  // Load employee if scoped
  if (!trigger[0].agent_employee_id) return; // org-level triggers handled elsewhere
  
  const emp = await db.select().from(agentEmployees)
    .where(and(eq(agentEmployees.id, trigger[0].agent_employee_id), eq(agentEmployees.is_active, true)))
    .limit(1);
  if (!emp[0]) return;

  // Check daily limit
  if (emp[0].daily_action_count >= emp[0].max_daily_actions) return;

  const org = await db.select().from(orgs).where(eq(orgs.id, orgId)).limit(1);

  // Evaluate condition (structured field matching)
  const condition = trigger[0].condition as Record<string, any> | null;
  if (condition) {
    for (const [field, expected] of Object.entries(condition)) {
      const actual = eventData[field];
      if (Array.isArray(expected)) {
        if (!expected.includes(actual)) return; // condition not met
      } else {
        if (actual !== expected) return; // condition not met
      }
    }
  }

  // Execute: run agent with trigger's action config
  const actions = trigger[0].actions as { type: string; prompt?: string; tool?: string; params?: any }[];
  
  for (const action of actions) {
    if (action.type === 'run_agent' && action.prompt) {
      await runAgentQuery({
        content: action.prompt.replace('{{event}}', JSON.stringify(eventData)),
        orgId,
        userId: emp[0].user_id,
        orgName: org[0]?.name || '',
        mode: 'background',
        systemPromptOverride: emp[0].system_prompt,
        agentEmployeeId: emp[0].id,
      });
    }
  }

  // Update trigger stats
  await db.update(triggers).set({
    last_fired_at: new Date(),
    fire_count: (trigger[0].fire_count || 0) + 1,
  } as any).where(eq(triggers.id, triggerId));
}
```

Note: The triggers table may not have `last_fired_at` and `fire_count` columns. Check and add if needed, or skip the stats update.

// In the first catch block, before retrying:
